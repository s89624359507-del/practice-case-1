1. Что нужно было сделать
По заданию нужно было написать программу, которая:
1.	Последовательно спрашивает у пользователя день, месяц и год рождения.
2.	Определяет день недели для этой даты.
3.	Проверяет, был ли год високосным.
4.	Считает, сколько пользователю сейчас лет.
5.	Выводит дату рождения в нормальном формате. В задании фраза обрывается на «в формате д…», я поняла это как дд.мм.гггг.
Честно, сначала меня немного смутила формулировка «стилистическое преобразование чисел», но по пунктам уже было понятно, что работа идёт с датой. Поэтому я делала обычное консольное приложение на Node.js.
2. Ход работы
Сначала я хотела всё сделать через switch, но потом поняла, что для дня недели проще использовать стандартный объект Date. Для високосного года я написала отдельную функцию с классическим условием: год делится на 4, но не на 100, кроме случаев, когда делится на 400.
Возраст я считала не просто как «текущий год минус год рождения», а с проверкой месяца и дня. Иначе, например, если день рождения ещё не наступил в этом году, возраст получался бы на год больше.
Ввод я сделала через readline, потому что это консольное приложение на Node.js. Валидацию сильно не усложняла — проверила только, что введены числа. Я поняла, что идеально обрабатывать все ошибки вроде 31 февраля не стала, потому что для практики главное было реализовать саму логику.
3. Код программы
Файл date.js:
javascript
const readline = require('readline');

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout
});

function ask(question) {
  return new Promise((resolve) => {
    rl.question(question, (answer) => resolve(answer.trim()));
  });
}

function getDayOfWeek(day, month, year) {
  const date = new Date(year, month - 1, day);
  const days = [
    'воскресенье',
    'понедельник',
    'вторник',
    'среда',
    'четверг',
    'пятница',
    'суббота'
  ];

  return days[date.getDay()];
}

function isLeapYear(year) {
  return (year % 4 === 0 && year % 100 !== 0) || year % 400 === 0;
}

function calculateAge(day, month, year) {
  const today = new Date();

  let age = today.getFullYear() - year;
  const currentMonth = today.getMonth() + 1;
  const currentDay = today.getDate();

  if (
    currentMonth < month ||
    (currentMonth === month && currentDay < day)
  ) {
    age--;
  }

  return age;
}

function formatDate(day, month, year) {
  const dd = String(day).padStart(2, '0');
  const mm = String(month).padStart(2, '0');

  return `${dd}.${mm}.${year}`;
}

async function main() {
  const day = Number(await ask('Введите день рождения: '));
  const month = Number(await ask('Введите месяц рождения: '));
  const year = Number(await ask('Введите год рождения: '));

  if (!day || !month || !year) {
    console.log('Некорректная дата');
    rl.close();
    return;
  }

  console.log('\nРезультат:');
  console.log('Дата рождения:', formatDate(day, month, year));
  console.log('День недели:', getDayOfWeek(day, month, year));
  console.log('Високосный год:', isLeapYear(year) ? 'да' : 'нет');
  console.log('Возраст:', calculateAge(day, month, year));

  rl.close();
}

main();
